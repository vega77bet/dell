cmake_minimum_required(VERSION 3.22)
project(DellOS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Svg Test)
find_package(X11 REQUIRED)
find_package(qtermwidget6 REQUIRED)
add_library(dell-common STATIC apps/common.cpp apps/fileops.cpp)
target_link_libraries(dell-common PUBLIC Qt6::Widgets Qt6::Svg)
target_include_directories(dell-common PUBLIC apps)
add_executable(dell-app apps/windows.h apps/main.cpp apps/text.cpp apps/paint.cpp apps/files.cpp apps/terminal.cpp apps/media.cpp apps/capture.cpp apps/settings.cpp)
target_link_libraries(dell-app PRIVATE dell-common Qt6::Multimedia Qt6::MultimediaWidgets qtermwidget6)
add_executable(dell-shell desktop/main.cpp desktop/x11.cpp)
target_link_libraries(dell-shell PRIVATE dell-common X11::X11)
install(TARGETS dell-app dell-shell RUNTIME DESTINATION bin)
install(PROGRAMS os/dell-session DESTINATION bin)
install(DIRECTORY os/overlay/ DESTINATION / FILE_PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
enable_testing()
add_executable(dell-tests apps/windows.h tests/test_apps.cpp apps/text.cpp apps/paint.cpp)
target_link_libraries(dell-tests PRIVATE dell-common Qt6::Test)
add_test(NAME native-apps COMMAND dell-tests)
set_tests_properties(native-apps PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(dell-screenshot tests/screenshot.cpp)
target_link_libraries(dell-screenshot PRIVATE Qt6::Widgets)
